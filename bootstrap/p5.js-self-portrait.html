I used the program Processing to code and put together shapes and lines to form a more abstract self-portrait.

function setup() {
  createCanvas(600, 600);

  background(0, 255, 81);


fill(0)
ellipse(300, 300, 410, 380);
strokeWeight(0);
describe('hair outline.');

fill(0)
rect(94, 300, 411, 310);
strokeWeight(0);
describe('hair outline.');

fill(156, 0, 255)
ellipse(300, 300, 400, 370);
strokeWeight(0);
describe('purple circle part of hair.');

fill(156, 0, 255)
rect(100, 300, 399, 310);
strokeWeight(0);
describe('purple rectangle part of the long hair.');

fill(0)
ellipse(300, 300, 350, 350);
describe('black circle part of hair.');

fill(0)
rect(125, 300, 350, 300);
describe('black rectangle part of the long hair.');

fill(255)
ellipse(300, 300, 300, 320);
describe('white circle as the face.');

fill(0)
rect(180, 270, 90, 10)
describe('black rectangle for eyelids left.');

fill(0)
rect(180, 270, 30, 35)
describe('black pupil left.');

fill(0)
rect(330, 270, 90, 10)
describe('black rectangle for eyelids right.');

fill(0)
rect(330, 270, 30, 35)
describe('black pupil right.');

fill(0)
rect(260, 400, 70, 10)
describe('mouth');

strokeWeight(5)
stroke(255)
fill(0)
ellipse(300, 620, 300, 250);//body

fill(255)
rect(270, 455, 55, 50)
describe('neck');

}
