function levelOne(){
  text("Level 1", width/2, height-20);
  var distToBall= dist(ballx, bally, mouseX, mouseY);
  if(distToBall < ballSize/2){
    ballx=random(width);
    bally=random(height);
    score = score + 1;
  }// end of if stmt
  
  if(score>=5){ 
    gameState = "L2";
  }
   
  image(img, ballx-ballSize/2, bally-ballSize/2, ballSize, ballSize);
  line(ballx, bally, mouseX, mouseY);
  
} //end level 1
